const sheets = [...shadowRoot.querySelectorAll('.diary-sheet')];

const chapterSelect = shadowRoot.querySelector('.diary-select');
const pageJump = shadowRoot.querySelector('.diary-jump');
const thumbnails = shadowRoot.querySelector('.diary-thumbnails');
const thumbnailRange = shadowRoot.querySelector('.diary-thumb-range');
const previous = shadowRoot.querySelector('.diary-prev');
const next = shadowRoot.querySelector('.diary-next');

function isMobile() {
return window.matchMedia('(max-width: 680px)').matches;
}

function pageMarkup(page, index) {
if (!page) {
return '<div class="diary-sheet-content"><div class="diary-photo-wrap"><p class="diary-empty">这一章还没有日记照片。</p></div></div>';
}

const title = page.title ? '<h3 class="diary-page-title">' + page.title + '</h3>' : '';
const date = page.date ? '<span>' + page.date + '</span>' : '';
const note = page.note ? '<p class="diary-page-note">' + page.note + '</p>' : '';
return '<div class="diary-sheet-content"><div class="diary-page-label"><span>第 ' + (index + 1) + ' 页</span>' + date + '</div>' + title + '<div class="diary-photo-wrap"><img class="diary-photo" src="' + page.image + '" alt="罗德岛日记第 ' + (index + 1) + ' 页" fetchpriority="high"></div>' + note + '</div>';
}

function renderSheet(sheet, page, index, visible) {
sheet.classList.toggle('is-current', visible);
sheet.innerHTML = pageMarkup(page, index);
}

function thumbnailPageMarkup(page, index) {
const image = page ? (page.thumb || page.image) : '';
return '<span class="diary-thumb-page"><img src="' + image + '" alt="第 ' + (index + 1) + ' 页"><span class="diary-thumb-page-label">第 ' + (index + 1) + ' 页</span></span>';
}

function render() {
const diaryPages = chapters[currentChapterIndex] ? chapters[currentChapterIndex].pages : [];
const step = isMobile() ? 1 : 2;
const firstPage = diaryPages[currentIndex];
const secondPage = isMobile() ? null : diaryPages[currentIndex + 1];
renderSheet(sheets[0], firstPage, currentIndex, true);
renderSheet(sheets[1], secondPage, currentIndex + 1, !isMobile() && Boolean(secondPage));
count.textContent = diaryPages.length ? '第 ' + (currentIndex + 1) + (secondPage ? ' - ' + (currentIndex + 2) : '') + ' / ' + diaryPages.length + ' 页' : '等待上传日记照片';
previous.disabled = currentIndex === 0;
next.disabled = !diaryPages.length || currentIndex + step >= diaryPages.length;
const totalSpreads = Math.ceil(diaryPages.length / 2);
const currentSpread = Math.floor(currentIndex / 2);
thumbnailRange.textContent = diaryPages.length ? '每张缩略图代表 2 页 · 共显示 ' + diaryPages.length + ' 页' : '';
thumbnails.innerHTML = Array.from({ length: totalSpreads }, function (_, spread) {
const firstPage = diaryPages[spread * 2];
const secondPage = diaryPages[spread * 2 + 1];
const images = thumbnailPageMarkup(firstPage, spread * 2) + (secondPage ? thumbnailPageMarkup(secondPage, spread * 2 + 1) : '');
return '<button class="diary-thumb' + (spread === currentSpread ? ' is-active' : '') + '" type="button" title="第 ' + (spread * 2 + 1) + (secondPage ? ' - ' + (spread * 2 + 2) : '') + ' 页" aria-label="第 ' + (spread * 2 + 1) + (secondPage ? ' 到第 ' + (spread * 2 + 2) : '') + ' 页"><span class="diary-thumb-pages">' + images + '</span></button>';
}).join('');
thumbnails.querySelectorAll('.diary-thumb').forEach(function (button, offset) {
button.addEventListener('click', function () {
currentIndex = offset * 2;
render();
});
});
}

function setChapter(index) {
currentChapterIndex = index;
currentIndex = 0;
render();
}

function populateChapters() {
chapterSelect.innerHTML = chapters.map(function (chapter, index) {
return '<option value="' + index + '">' + chapter.title + '（' + chapter.pages.length + '页）</option>';
}).join('');
chapterSelect.value = String(currentChapterIndex);
}

previous.addEventListener('click', function () {
currentIndex = Math.max(0, currentIndex - (isMobile() ? 1 : 2));
render();
});

next.addEventListener('click', function () {
const diaryPages = chapters[currentChapterIndex] ? chapters[currentChapterIndex].pages : [];
currentIndex = Math.min(diaryPages.length - 1, currentIndex + (isMobile() ? 1 : 2));
render();
});

chapterSelect.addEventListener('change', function () {
setChapter(Number(chapterSelect.value));
});

pageJump.addEventListener('change', function () {
const diaryPages = chapters[currentChapterIndex] ? chapters[currentChapterIndex].pages : [];
const requested = Number(pageJump.value) - 1;
if (Number.isInteger(requested) && requested >= 0 && requested < diaryPages.length) {
currentIndex = isMobile() ? requested : requested - (requested % 2);
render();
}
pageJump.value = '';
});

document.addEventListener('keydown', function (event) {
if (!host.isConnected) return;
if (event.key === 'ArrowLeft') previous.click();
if (event.key === 'ArrowRight') next.click();
});

window.addEventListener('resize', render);

populateChapters();
render();

})();
</script>
